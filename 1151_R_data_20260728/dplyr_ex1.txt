########################

flights	%>%

flights	%>%

flights%>%

flights%>%

d1=flights%>%

#  2. gain>0

d2=flights%>%

## flights 中 計算出 每個tailnum的飛行次數，平均飛行距離，平均延遲抵達時間，＃#

flights %>% 
  group_by(tailnum)%>%
  summarise(
    count=n(),
    dist=mean(distance,na.rm=T),
    delay=mean(arr_delay,na.rm=T)
  )%>%
  filter(is.finite(delay)) %>%
  arrange( desc(count))
  

## flights 中 計算出 每個dest中 總共的飛機(tailnum)個數，總共的飛行次數，再根據飛行次數排序

flights %>%
  group_by(dest)%>%
  summarise(
    plane=n_distinct(tailnum),
    flights=n()
  )%>%
  arrange(desc(flights))

## 1. 找出 每輛飛機的 飛行次數，平均飛行距離，平均delay時間  ##
## 2. 並依此畫出 飛行距離 和 delay 時間, 再找出好的呈現方式

delay_data=flights%>%
  group_by(tailnum)%>%
  summarise(count=n(),
            avg_dist=mean(distance,na.rm=T),
            avg_delay=mean(arr_delay,na.rm=T))%>%
  filter(count>20,avg_dist<2000)


ggplot(data=delay_data,aes(avg_dist,avg_delay))+
  geom_point()

ggplot(data=delay_data,aes(avg_dist,avg_delay))+
  geom_point(aes(size=count),alpha=.5)

ggplot(data=delay_data,aes(avg_dist,avg_delay))+
  geom_point(aes(size=count),alpha=.3)+
  geom_smooth()

ggplot(data=delay_data,aes(avg_dist,avg_delay))+
  geom_point(aes(size=count),alpha=.3)+
  geom_smooth()+
  scale_size_area()
