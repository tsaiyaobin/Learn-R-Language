pl<-ggplot(data=mtcars,aes(x=mpg,y=hp))

pl+geom_point()
pl+geom_point()+geom_smooth()
pl+geom_point()+geom_quantile()

pl+geom_point()+facet_grid(cyl~.)
## Add statistics
pl+geom_point()+facet_grid(cyl~.)+geom_smooth()
## Add coordinates
pl+geom_point()+facet_grid(cyl~.)+
  stat_smooth()+coord_cartesian(xlim=c(15,25))

pl+geom_point()+facet_grid(cyl~.)+
  stat_smooth()+coord_cartesian(xlim=c(15,25))+
  theme_dark()

#different cylinders and size is set to 5 commonly
#for all points.

                  color=factor(cyl)),
              size=5)
